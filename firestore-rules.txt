FIRESTORE SECURITY RULES for Quote Vault (with admin + maintenance mode)
=========================================================================

HOW TO APPLY (one time, ~1 minute):

1. Go to https://console.firebase.google.com
2. Pick the "quotevault-faa66" project
3. Left menu: Build -> Firestore Database -> Rules
4. Delete everything in the editor, paste ALL the rules below, click Publish

These rules add:
- An admin account (lpbmusic5426@gmail.com) that can read ALL data
- A config collection for maintenance mode (anyone can read, admin can write)

----- COPY EVERYTHING BELOW THIS LINE -----

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isSignedIn() { return request.auth != null; }
    function isAdmin() {
      return isSignedIn()
        && (request.auth.token.email == "lpbmusic5426@gmail.com"
            || request.auth.token.email == "lpbyars@gmail.com");
    }
    function isGroupOwner(groupId) {
      return isSignedIn()
        && get(/databases/$(database)/documents/groups/$(groupId)).data.ownerUid == request.auth.uid;
    }
    function isGroupMember(groupId) {
      return isSignedIn()
        && exists(/databases/$(database)/documents/groups/$(groupId)/members/$(request.auth.uid));
    }

    // Personal quotes — owner or admin can read/write.
    match /quotes/{quoteId} {
      allow get, list: if isSignedIn() && (resource.data.uid == request.auth.uid || isAdmin());
      allow create: if isSignedIn() && request.resource.data.uid == request.auth.uid;
      allow update, delete: if isSignedIn() && (resource.data.uid == request.auth.uid || isAdmin());
    }

    // Config collection — maintenance mode flag etc.
    // Anyone (even unsigned-in) can read so the app can check maintenance
    // status before sign-in. Only admin can write.
    match /config/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Membership pointers.
    match /users/{uid}/groupMemberships/{groupId} {
      allow read, write: if isSignedIn() && (request.auth.uid == uid || isGroupOwner(groupId));
    }

    // Groups — signed-in users can read; owner or admin can change.
    match /groups/{groupId} {
      allow read: if isSignedIn();
      allow create: if isSignedIn() && request.resource.data.ownerUid == request.auth.uid;
      allow update, delete: if isSignedIn() && (isGroupOwner(groupId) || isAdmin());
    }

    // Group members.
    match /groups/{groupId}/members/{uid} {
      allow read: if isSignedIn();
      allow create, update: if isSignedIn() && request.auth.uid == uid;
      allow delete: if isSignedIn() && (request.auth.uid == uid || isGroupOwner(groupId) || isAdmin());
    }

    // Shared group quotes — members or admin can read; author or admin can delete.
    match /groups/{groupId}/quotes/{quoteId} {
      allow get, list: if isGroupMember(groupId) || isAdmin();
      allow create: if isGroupMember(groupId) && request.resource.data.uid == request.auth.uid;
      allow update, delete: if isSignedIn() && (resource.data.uid == request.auth.uid || isAdmin()) && isGroupMember(groupId);
    }
  }
}
