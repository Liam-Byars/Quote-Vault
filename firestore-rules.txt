FIRESTORE SECURITY RULES for Quote Vault (shared groups + real-time)
=====================================================================

HOW TO APPLY (one time, ~1 minute):

1. Go to https://console.firebase.google.com
2. Pick the "quotevault-faa66" project
3. Left menu: Build -> Firestore Database -> Rules
4. Delete everything in the editor, paste ALL the rules below, click Publish

WHY: the app now stores group quotes in a shared spot
(groups/{groupId}/quotes) so everyone in a group sees new quotes within
seconds. The old rules blocked those writes, which made creating groups
and sharing quotes fail.

----- COPY EVERYTHING BELOW THIS LINE -----

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Personal quotes — only you can read/write your own.
    match /quotes/{quoteId} {
      allow read, update, delete: if request.auth != null
        && resource.data.uid == request.auth.uid;
      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid;
    }

    // Groups — signed-in users can read (needed to look up invite
    // codes); only the owner can change or delete a group.
    match /groups/{groupId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null
        && request.resource.data.ownerUid == request.auth.uid;
      allow update, delete: if request.auth != null
        && resource.data.ownerUid == request.auth.uid;

      // Joined members — one doc per user, doc id = their uid.
      match /members/{memberId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null
          && request.auth.uid == memberId;
        allow delete: if request.auth != null
          && (request.auth.uid == memberId
              || get(/databases/$(database)/documents/groups/$(groupId)).data.ownerUid == request.auth.uid);
      }

      // Shared group quotes — members see them in real time.
      match /quotes/{quoteId} {
        function isMember() {
          return request.auth != null
            && exists(/databases/$(database)/documents/users/$(request.auth.uid)/groupMemberships/$(groupId));
        }
        function isGroupOwner() {
          return request.auth != null
            && get(/databases/$(database)/documents/groups/$(groupId)).data.ownerUid == request.auth.uid;
        }
        allow read: if isMember();
        allow create: if isMember()
          && request.resource.data.uid == request.auth.uid;
        allow update: if isMember();
        allow delete: if isMember()
          && (resource.data.uid == request.auth.uid || isGroupOwner());
      }
    }

    // Membership pointers — your own list; the group owner cleans them
    // up when deleting a group.
    match /users/{uid}/groupMemberships/{groupId} {
      allow read, write: if request.auth != null
        && (request.auth.uid == uid
            || get(/databases/$(database)/documents/groups/$(groupId)).data.ownerUid == request.auth.uid);
    }
  }
}
